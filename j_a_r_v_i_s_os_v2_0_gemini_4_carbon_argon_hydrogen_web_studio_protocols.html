Готовый C++ скетч для Arduino Uno 
#include <Servo.h>

Servo jarvisServo;
const int PIN_RELAY = 7;
const int PIN_SERVO = 9;
const int PIN_TRIG  = 10;
const int PIN_ECHO  = 11;
String cmd = "";

long readRadarCm() {
  digitalWrite(PIN_TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(PIN_TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(PIN_TRIG, LOW);
  long dur = pulseIn(PIN_ECHO, HIGH, 25000);
  return dur == 0 ? 120 : dur * 0.034 / 2;
}

void setup() {
  Serial.begin(115200);
  pinMode(PIN_RELAY, OUTPUT);
  pinMode(PIN_TRIG, OUTPUT);
  pinMode(PIN_ECHO, INPUT);
  jarvisServo.attach(PIN_SERVO);
  jarvisServo.write(90);
}

void loop() {
  while (Serial.available() > 0) {
    char c = Serial.read();
    if (c == '\n') {
      cmd.trim();
      if (cmd == "RELAY:1") digitalWrite(PIN_RELAY, HIGH);
      else if (cmd == "RELAY:0") digitalWrite(PIN_RELAY, LOW);
      else if (cmd.startsWith("SERVO:")) {
        int ang = cmd.substring(6).toInt();
        jarvisServo.write(constrain(ang, 0, 180));
      }
      long cm = readRadarCm();
      Serial.print("{\"cmd\":\""); Serial.print(cmd);
      Serial.print("\",\"cm\":"); Serial.print(cm);
      Serial.println("}");
      cmd = "";
    } else {
      cmd += c;
    }
  }
}

Как работает каскад Gemini 4 Carbon / Argon / Hydrogen на твоих устройствах

Шаг 1. Умный каскад Gemini 4

Выбери gemini-4-carbon в верхнем меню и подключи свой GEMINI_API_KEY. Если на ключе активен Paid Tier / бонус Pro, ответит напрямую Carbon; если ключ на Free Tier — система за 50 мс автоматически переключит запрос на бесплатную gemini-4-hydrogen без единой ошибки!

Шаг 2. Сайты, Протоколы и Анализы

Ты можешь писать прямо в основной чат: «Джарвис, создай сайт-тренажёр», «Активируй Протокол Олимпиада» или «Сделай анализ трикоптера» — я сам вызову нужный инструмент и переключу тебя на готовую вкладку с результатом.

Шаг 3. Выложи на GitHub Pages / Vercel
